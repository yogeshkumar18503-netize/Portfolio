library_system.py 
class LibrarySystem:
    def __init__(self):
        self.inventory = {}
        self.borrowed = {}
        self.daily_rate = 5.0  # Daily penalty in INR

    def calculate_rental_charge(self, days_held, grace_period=14):
        # Automated fee calculation algorithm
        if days_held <= grace_period:
            return 0.0
        overdue = days_held - grace_period
        return overdue * self.daily_rate

    def return_book(self, member_id, book_id, days):
        charge = self.calculate_rental_charge(days)
        self.inventory[book_id]['status'] = 'AVAILABLE'
        return {'status': 'Success', 'fee': charge}

-- Venue Schedule & Student Record Query
SELECT 
    e.event_id,
    e.event_name,
    v.venue_name,
    COUNT(s.student_id) AS total_participants,
    e.start_time,
    e.end_time
FROM events e
JOIN venues v ON e.venue_id = v.id
LEFT JOIN student_attendance s ON e.event_id = s.event_id
GROUP BY e.event_id
ORDER BY e.start_time ASC;
